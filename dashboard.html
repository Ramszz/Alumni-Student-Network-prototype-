    Aptitude Practice Zone

    Try answering the questions first from the “Questions” folder. If you get stuck, open the “Answers” folder with the same question numbers.

    Aptitude Questions (Sample – extend to 100+)
    1. Q1. A train travels at 60 km/h for 2.5 hours. How far does it travel?
    2. Q2. Find ¾ of 2/5 of 200.
    3. Q3. A number becomes 180 when increased by 20%. What is the original number?
    4. Q4. Simple interest on ₹5000 at 8% per annum for 2 years is how much?
    5. Q5. Divide ₹600 between A and B in the ratio 2:3. How much does B get?
    6. Q6. Find the average of 5, 7, 9, 11, 13.
    7. Q7. A can finish a work in 10 days, B in 15 days. In how many days will they finish together?
    8. Q8. If 15% of a number is 90, what is the number?
    9. Q9. Successive discounts of 10% and 20% are given on a marked price of ₹1000. What is the final selling price?
    10. Q10. Find the missing term in the series: 2, 6, 12, 20, 30, ?
    Aptitude Answers (for Questions Above)
    1. A1. Distance = 60 × 2.5 = 150 km.
    2. A2. (3/4) × (2/5) × 200 = 60.
    3. A3. 1.2x = 180 ⇒ x = 150.
    4. A4. SI = (5000 × 8 × 2)/100 = ₹800.
    5. A5. B gets (3/5) × 600 = ₹360.
    6. A6. Average = 45/5 = 9.
    7. A7. 1/10 + 1/15 = 5/30 = 1/6 ⇒ 6 days.
    8. A8. 0.15x = 90 ⇒ x = 600.
    9. A9. After 10%: 1000 × 0.9 = 900; then 900 × 0.8 = ₹720.
    10. A10. Differences +4, +6, +8, +10, +12 ⇒ next = 30 + 12 = 42.

int main() {
    int n, i, x;
    long long sum = 0;
    printf("Enter count: ");
    scanf("%d", &n);

    2. C-A2. 
      int isPrime(int n) {
    if (n <= 1) return 0;
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) return 0;
    }
    return 1;
}
    3. C-A3. 
      int reverse(int n) {
    int rev = 0;
    while (n != 0) {
        int d = n % 10;
        rev = rev * 10 + d;
        n /= 10;
    }
    return rev;
}
    4. C-A4. 
      int countVowels(const char *s) {
    int count = 0;
    for (int i = 0; s[i] != '\0'; i++) {
        char ch = s[i];
        if (ch=='a'||ch=='e'||ch=='i'||ch=='o'||ch=='u'||
            ch=='A'||ch=='E'||ch=='I'||ch=='O'||ch=='U') {
            count++;
        }
    }
    return count;
}
    5. C-A5. 
      int maxOfThree(int a, int b, int c) {
    int max = a;
    if (b > max) max = b;
    if (c > max) max = c;
    return max;
}
    6. C-A6. 
      long long fact(int n) {
    long long f = 1;
    for (int i = 1; i <= n; i++) {
        f *= i;
    }
    return f;
}

    4. CPP-A4. 
      int secondLargest(const vector<int>& r) {
    if (r.size() < 2) return -1;
    int max1 = INT_MIN, max2 = INT_MIN;
    for (int x : r) {
        if (x > max1) {
            max2 = max1;
            max1 = x;
        } else if (x > max2 && x != max1) {
            max2 = x;
        }
    }
    return max2;
}
    5. CPP-A5. 
      long long power(long long a, int b) {
    if (b == 0) return 1;
    if (b % 2 == 0) {
        long long half = power(a, b / 2);
        return half * half;
    }
    return a * power(a, b - 1);
}
    6. CPP-A6. 
      class Stack {
    int arr[100];
    int top;
public:
    Stack() { top = -1; }

    bool push(int x) {
        if (top == 99) return false;
        arr[++top] = x;
        return true;
    }

    bool pop(int &x) {
        if (top == -1) return false;
        x = arr[top--];
        return true;
    }

    bool isEmpty() const { return top == -1; }
};
